cmake_minimum_required(VERSION 3.22)
project(moqlab_simulation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ns3 REQUIRED PATHS "${CMAKE_CURRENT_SOURCE_DIR}/.build/ns3-install/lib/cmake/ns3" NO_DEFAULT_PATH)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
set(PICO "${CMAKE_CURRENT_SOURCE_DIR}/.build/picoquic")
set(PICO_BUILD "${CMAKE_CURRENT_SOURCE_DIR}/.build/pico-build")
set(PTLS "${PICO_BUILD}/_deps/picotls-src")
set(PTLS_BUILD "${PICO_BUILD}/_deps/picotls-build")
add_executable(moqlab-sim native/main.cpp ../../src/sbd/Detector.cpp ../../src/sbd/DcSbd.cpp ../../src/sbd/ReceiveIntervals.cpp)
target_include_directories(moqlab-sim PRIVATE ../../src .build/include "${PICO}/picoquic" "${PTLS}/include")
target_link_libraries(moqlab-sim PRIVATE ns3::internet ns3::bridge ns3::traffic-control ns3::applications
"-Wl,--start-group" "${PICO_BUILD}/libpicoquic-core.a" "${PICO_BUILD}/libpicoquic-log.a" "-Wl,--end-group"
"${PTLS_BUILD}/libpicotls-openssl.a" "${PTLS_BUILD}/libpicotls-minicrypto.a" "${PTLS_BUILD}/libpicotls-fusion.a" "${PTLS_BUILD}/libpicotls-core.a"
OpenSSL::SSL OpenSSL::Crypto Threads::Threads ${CMAKE_DL_LIBS})
target_compile_options(moqlab-sim PRIVATE -Wall -Wextra -Wno-unused-parameter)
add_executable(sbd-check native/test_core.cpp ../../src/sbd/Detector.cpp ../../src/sbd/DcSbd.cpp ../../src/sbd/ReceiveIntervals.cpp)
target_include_directories(sbd-check PRIVATE ../../src)
enable_testing()
add_test(NAME sbd-check COMMAND sbd-check)
add_test(NAME wire-check COMMAND moqlab-sim --check-wire)
add_executable(sbd-replay native/replay.cpp ../../src/sbd/Detector.cpp ../../src/sbd/DcSbd.cpp ../../src/sbd/ReceiveIntervals.cpp)
target_include_directories(sbd-replay PRIVATE ../../src .build/include)
